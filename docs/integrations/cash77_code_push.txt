// Cashback One — отправка кода списания бонусов в приложение магазина
// Для POSCENTER Cash (1С:Предприятие 7.7). Замена отправки через WhatsApp.
//
// Как работает: касса сама придумывает 4-значный код, вызывает
// ОтправитьКодЧерезCashbackOne(Телефон, Код). Облако Cashback One находит
// покупателя с картой этого магазина и присылает ему push-уведомление
// «Ваш код для списания бонусов: 1234…» в приложение магазина.
// Покупатель называет код кассиру, касса сама сравнивает его со своим.
//
// Возврат: 1 — уведомление отправлено, 0 — ошибка (кассиру показано
// сообщение). При 0 списание не проводить.
//
// КлючКассы выпускается в кабинете Cashback One (страница магазина,
// раздел «Кассы»), начинается с "pos_". Ключ хранить в настройках Cash,
// не в тексте модуля и не показывать кассиру.
//
// Требуется метод POST /pos/code-push на сервере Cashback One.
// Документация: docs/POS_REDEEM_API.md, раздел /pos/code-push.
// =====================================================================

// Случайный 4-значный код "0000"–"9999". Каждый раз новый, не постоянный.
Функция СлучайныйКодCashbackOne() Экспорт
	Код = "";
	Попытка
		Скрипт = СоздатьОбъект("MSScriptControl.ScriptControl");
		Скрипт.Language = "VBScript";
		Скрипт.ExecuteStatement("Randomize");
		Число = Скрипт.Eval("Int(Rnd() * 10000)");
		Код = СокрЛП(Строка(Число));
	Исключение
		// Запасной вариант, если ScriptControl недоступен
		Код = СокрЛП(Строка(Цел(ТекущееВремя() * 7919 % 10000)));
	КонецПопытки;
	Пока СтрДлина(Код) < 4 Цикл
		Код = "0" + Код;
	КонецЦикла;
	Возврат Код;
КонецФункции

// Достаёт код ошибки из ответа вида {"error":{"code":"CARD_NOT_FOUND",...}}
Функция КодОшибкиCashbackOne(Ответ)
	Кавычка = Симв(34);
	Метка = Кавычка + "code" + Кавычка + ":" + Кавычка;
	Поз = Найти(Ответ, Метка);
	Если Поз = 0 Тогда
		Возврат "";
	КонецЕсли;
	Хвост = Сред(Ответ, Поз + СтрДлина(Метка));
	Конец = Найти(Хвост, Кавычка);
	Если Конец = 0 Тогда
		Возврат "";
	КонецЕсли;
	Возврат Лев(Хвост, Конец - 1);
КонецФункции

Функция ОтправитьКодЧерезCashbackOne(НомерПолучателя, Код) Экспорт
	КлючКассы = "pos_ВСТАВЬТЕ_КЛЮЧ_КАССЫ";   // взять из настроек Cash
	URL = "https://api.cashback-qr.poscenter.kz/pos/code-push";

	НомерПолучателя = СокрЛП(НомерПолучателя);
	НомерПолучателя = СтрЗаменить(НомерПолучателя, "+", "");
	НомерПолучателя = СтрЗаменить(НомерПолучателя, " ", "");
	НомерПолучателя = СтрЗаменить(НомерПолучателя, "-", "");
	НомерПолучателя = СтрЗаменить(НомерПолучателя, "(", "");
	НомерПолучателя = СтрЗаменить(НомерПолучателя, ")", "");
	Если СтрДлина(НомерПолучателя) <> 11 Тогда
		Предупреждение("Неверный номер телефона покупателя.");
		Возврат 0;
	КонецЕсли;
	Код = СокрЛП(Код);
	Если СтрДлина(Код) <> 4 Тогда
		Предупреждение("Код должен состоять из 4 цифр.");
		Возврат 0;
	КонецЕсли;

	Попытка
		Кавычка = Симв(34);
		// В теле только цифры — экранирование не нужно
		JSON = "{" +
		Кавычка + "phone" + Кавычка + ":" + Кавычка + НомерПолучателя + Кавычка + "," +
		Кавычка + "code"  + Кавычка + ":" + Кавычка + Код + Кавычка +
		"}";

		HTTP = СоздатьОбъект("MSXML2.XMLHTTP.6.0");
		HTTP.Open("POST", URL, 0);
		HTTP.setRequestHeader("Content-Type", "application/json");
		HTTP.setRequestHeader("X-Pos-Key", КлючКассы);
		HTTP.Send(JSON);

		КодОтвета = HTTP.Status;
		Ответ = HTTP.responseText;
	Исключение
		Предупреждение("Нет связи с Cashback One. Проверьте интернет и попробуйте ещё раз.");
		Возврат 0;
	КонецПопытки;

	Если КодОтвета = 200 Тогда
		Возврат 1;
	КонецЕсли;

	Ошибка = КодОшибкиCashbackOne(Ответ);
	Если Ошибка = "CARD_NOT_FOUND" Тогда
		Предупреждение("Карта с этим номером телефона не найдена в магазине.");
	ИначеЕсли Ошибка = "CARD_BLOCKED" Тогда
		Предупреждение("Карта покупателя заблокирована. Списание невозможно.");
	ИначеЕсли Ошибка = "CUSTOMER_NO_APP" Тогда
		Предупреждение("У покупателя нет приложения магазина или выключены уведомления. Попросите установить приложение.");
	ИначеЕсли Ошибка = "RATE_LIMIT_EXCEEDED" Тогда
		Предупреждение("Слишком много запросов кода. Подождите несколько минут.");
	ИначеЕсли (Ошибка = "PUSH_DELIVERY_FAILED") ИЛИ (Ошибка = "PUSH_NOT_CONFIGURED") Тогда
		Предупреждение("Не удалось отправить код покупателю. Попробуйте ещё раз.");
	ИначеЕсли (Ошибка = "POS_KEY_MISSING") ИЛИ (Ошибка = "POS_KEY_INVALID") ИЛИ (Ошибка = "POS_KEY_REVOKED") Тогда
		Предупреждение("Неверный ключ кассы Cashback One. Обратитесь к администратору.");
	ИначеЕсли Ошибка = "VALIDATION_ERROR" Тогда
		Предупреждение("Неверный номер телефона или код.");
	Иначе
		Предупреждение("Ошибка Cashback One (HTTP " + КодОтвета + "). Попробуйте ещё раз.");
	КонецЕсли;
	Возврат 0;
КонецФункции
