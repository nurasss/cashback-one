// Cashback One / POSCENTER Cash / 1С:Предприятие 7.7
// Отправка кода списания бонусов покупателю в приложение магазина
// (замена отправки через WhatsApp). Метод сервера: POST /pos/code-push,
// документация — docs/POS_REDEEM_API.md.
//
// Касса сама генерирует код, отправляет его через Cashback One
// и сама сравнивает с кодом, названным покупателем.
// Возврат ОтправитьКодЧерезCashbackOne: 1 = HTTP 200, 0 = ошибка.
//
// ВАЖНО ПРИ ПЕРВОЙ ПРОВЕРКЕ НА КАССЕ:
// Используется MSXML2.ServerXMLHTTP.6.0 (есть тайм-ауты). Он НЕ берёт
// прокси из настроек Internet Explorer, а на старых Windows (7 и ниже)
// у него может быть выключен TLS 1.2 — тогда будет «Нет связи», хотя
// интернет есть. В таком случае заменить ServerXMLHTTP на
// MSXML2.XMLHTTP.6.0 (как в прежней WhatsApp-функции) и удалить строку
// HTTP.setTimeouts(...). Больше ничего менять не нужно.
// =====================================================================

// Проверяет, что строка состоит только из десятичных цифр.
Функция ТолькоЦифрыCashbackOne(Значение)
    Если СтрДлина(Значение) = 0 Тогда
        Возврат 0;
    КонецЕсли;

    Для Индекс = 1 По СтрДлина(Значение) Цикл
        Если Найти("0123456789", Сред(Значение, Индекс, 1)) = 0 Тогда
            Возврат 0;
        КонецЕсли;
    КонецЦикла;
    Возврат 1;
КонецФункции

// Генерирует строку из 4 цифр, включая ведущие нули.
// ScriptControl — старый генератор Rnd, НЕ криптографически стойкий.
// Запасной вариант по времени предсказуем и может давать совпадения.
Функция СлучайныйКодCashbackOne() Экспорт
    Код = "";
    Попытка
        Скрипт = СоздатьОбъект("MSScriptControl.ScriptControl");
        Скрипт.Language = "VBScript";
        Скрипт.ExecuteStatement("Randomize");
        ЧислоКода = Скрипт.Eval("Int(Rnd() * 10000)");
        Код = СокрЛП(Строка(ЧислоКода));
    Исключение
        // В 1С 7.7 ТекущееВремя() возвращает СТРОКУ.
        // Часы, минуты и секунды получаем через параметры как числа.
        ТекущееВремя(Часы, Минуты, Секунды);
        ЧислоКода = (Часы * 3600 + Минуты * 60 + Секунды) * 7919;
        ЧислоКода = ЧислоКода - Цел(ЧислоКода / 10000) * 10000;
        Код = СокрЛП(Строка(ЧислоКода));
    КонецПопытки;

    Пока СтрДлина(Код) < 4 Цикл
        Код = "0" + Код;
    КонецЦикла;
    Возврат Код;
КонецФункции

// Возвращает error.code из JSON; допускает пробелы, табы и переносы.
// Это ограниченный разбор известного формата API, не общий JSON-парсер.
Функция КодОшибкиCashbackOne(Знач Ответ)
    Кавычка = Симв(34);
    Ответ = СтрЗаменить(Ответ, " ", "");
    Ответ = СтрЗаменить(Ответ, Симв(9), "");
    Ответ = СтрЗаменить(Ответ, Симв(10), "");
    Ответ = СтрЗаменить(Ответ, Симв(13), "");

    Метка = Кавычка + "code" + Кавычка + ":" + Кавычка;
    Поз = Найти(Ответ, Метка);
    Если Поз = 0 Тогда
        Возврат "";
    КонецЕсли;

    Хвост = Сред(Ответ, Поз + СтрДлина(Метка));
    КонецКода = Найти(Хвост, Кавычка);
    Если КонецКода = 0 Тогда
        Возврат "";
    КонецЕсли;
    Возврат Лев(Хвост, КонецКода - 1);
КонецФункции

// Отправляет уже созданный в Cash код push-уведомлением покупателю.
// Сигнатура и результат сохранены, чтобы не менять код вызова в Cash.
Функция ОтправитьКодЧерезCashbackOne(Знач НомерПолучателя, Знач Код) Экспорт
    // В рабочей конфигурации брать ключ из защищённых настроек Cash.
    // Не включать настоящий ключ в исходный текст и журналы.
    КлючКассы = "pos_ВСТАВЬТЕ_КЛЮЧ_КАССЫ";
    URL = "https://api.cashback-qr.poscenter.kz/pos/code-push";

    Если (КлючКассы = "") ИЛИ (КлючКассы = "pos_ВСТАВЬТЕ_КЛЮЧ_КАССЫ") Тогда
        Предупреждение("Не задан ключ кассы Cashback One. Обратитесь к администратору.");
        Возврат 0;
    КонецЕсли;

    НомерПолучателя = СокрЛП(НомерПолучателя);
    НомерПолучателя = СтрЗаменить(НомерПолучателя, "+", "");
    НомерПолучателя = СтрЗаменить(НомерПолучателя, " ", "");
    НомерПолучателя = СтрЗаменить(НомерПолучателя, "-", "");
    НомерПолучателя = СтрЗаменить(НомерПолучателя, "(", "");
    НомерПолучателя = СтрЗаменить(НомерПолучателя, ")", "");
    НомерПолучателя = СтрЗаменить(НомерПолучателя, Симв(9), "");

    Если (СтрДлина(НомерПолучателя) <> 11) ИЛИ (ТолькоЦифрыCashbackOne(НомерПолучателя) = 0) Тогда
        Предупреждение("Неверный номер телефона покупателя. Нужно 11 цифр.");
        Возврат 0;
    КонецЕсли;

    Код = СокрЛП(Код);
    Если (СтрДлина(Код) <> 4) ИЛИ (ТолькоЦифрыCashbackOne(Код) = 0) Тогда
        Предупреждение("Код должен состоять из 4 цифр.");
        Возврат 0;
    КонецЕсли;

    Кавычка = Симв(34);
    // Здесь поля содержат только цифры, поэтому JSON-экранирование не требуется.
    JSON = "{" + Кавычка + "phone" + Кавычка + ":" + Кавычка + НомерПолучателя + Кавычка + "," + Кавычка + "code" + Кавычка + ":" + Кавычка + Код + Кавычка + "}";

    Попытка
        // ServerXMLHTTP позволяет выставить таймауты синхронного вызова.
        HTTP = СоздатьОбъект("MSXML2.ServerXMLHTTP.6.0");
        // DNS 3 с, соединение 5 с, отправка 5 с, получение 10 с.
        HTTP.setTimeouts(3000, 5000, 5000, 10000);
        HTTP.Open("POST", URL, 0);
        HTTP.setRequestHeader("Content-Type", "application/json; charset=utf-8");
        HTTP.setRequestHeader("X-Pos-Key", КлючКассы);
        HTTP.Send(JSON);
        КодОтвета = HTTP.Status;
        Ответ = HTTP.responseText;
    Исключение
        // После таймаута запрос МОГ дойти до сервера.
        // Не выполнять автоматическую повторную отправку.
        Предупреждение("Нет связи с Cashback One или истекло время ожидания. Проверьте интернет и состояние уведомления перед повтором.");
        Возврат 0;
    КонецПопытки;

    Если КодОтвета = 200 Тогда
        Возврат 1;
    КонецЕсли;

    Ошибка = КодОшибкиCashbackOne(Ответ);
    Если Ошибка = "CARD_NOT_FOUND" Тогда
        Предупреждение("Карта с этим номером телефона не найдена в магазине.");
    ИначеЕсли Ошибка = "CARD_BLOCKED" Тогда
        Предупреждение("Карта покупателя заблокирована. Списание невозможно.");
    ИначеЕсли Ошибка = "CUSTOMER_NO_APP" Тогда
        Предупреждение("У покупателя нет приложения магазина или активного устройства. Попросите установить приложение.");
    ИначеЕсли Ошибка = "RATE_LIMIT_EXCEEDED" Тогда
        Предупреждение("Слишком много запросов кода. Подождите несколько минут.");
    ИначеЕсли (Ошибка = "PUSH_DELIVERY_FAILED") ИЛИ (Ошибка = "PUSH_NOT_CONFIGURED") Тогда
        Предупреждение("Не удалось отправить код покупателю. Обратитесь к администратору, если ошибка повторяется.");
    ИначеЕсли (Ошибка = "POS_KEY_MISSING") ИЛИ (Ошибка = "POS_KEY_INVALID") ИЛИ (Ошибка = "POS_KEY_REVOKED") Тогда
        Предупреждение("Неверный или отозванный ключ кассы Cashback One. Обратитесь к администратору.");
    ИначеЕсли Ошибка = "VALIDATION_ERROR" Тогда
        Предупреждение("Неверный номер телефона или код.");
    Иначе
        Предупреждение("Ошибка Cashback One (HTTP " + СокрЛП(Строка(КодОтвета)) + "). Попробуйте ещё раз.");
    КонецЕсли;
    Возврат 0;
КонецФункции
